@throws() pub error_t read_count(borrowed FILE *file, borrowed mut int *out_count);
@throws(error) pub counter_t *create_counter(int initial, borrowed mut error_t *error);
typedef struct counter_t {
  @throws() pub error_t load(borrowed mut *self);
} counter_t;
void run(void) {
  @try {
    read_count(file, &count);
    counter = create_counter(count);
  }
  @catch (ERROR_ALLOCATION, error_t error) { report(error); }
  @catch (ERROR_INVALID_ARGUMENT | ERROR_IO, error_t error) { report(error); }
  @catch (error_t error) { report(error); }
}
