typedef struct counter_t {
  int value;
  borrowed mut char *buffer;
  pub int increment(borrowed mut *self, int amount) { self->value += amount; return 0; }
  static pub counter_t *create(int initial) { return 0; }
  priv int reset(borrowed mut *self);
  pub void render(borrowed const *self);
} counter_t;

unsigned char *allocator_cast(void *source) {
  return (warm unsigned char *)source;
}
